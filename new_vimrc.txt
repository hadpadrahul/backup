"==================================================
" OSC52 clipboard over SSH
"==================================================

" Copies yanks from remote Vim to local terminal clipboard.
" Requires terminal OSC52 support.

function! OSC52Copy()
    if !exists('v:event')
        return
    endif

    let l:content = join(v:event.regcontents, "\n")

    if empty(l:content)
        return
    endif

    let l:encoded = system(
        \ 'base64 | tr -d "\n"',
        \ l:content
        \ )

    let l:sequence =
        \ "\033]52;c;" . l:encoded . "\007"

    call system(
        \ 'printf "' .
        \ substitute(l:sequence, '"', '\\"', 'g') .
        \ '" > /dev/tty'
        \ )

endfunction


if exists('##TextYankPost')

    augroup OSC52
        autocmd!

        autocmd TextYankPost *
            \ call OSC52Copy()

    augroup END

endif

"==================================================
" Commenting (no plugins)
"==================================================

function! ToggleComment(start, end) abort

    if &commentstring == ''
        echo "No commentstring"
        return
    endif

    let cs = &commentstring

    let left = substitute(cs, '%s.*', '', '')
    let right = substitute(cs, '.*%s', '', '')

    let lines = getline(a:start, a:end)

    let all_commented = 1

    for line in lines
        if trim(line) !=# ''
            if line !~ '^\s*' . escape(left, '\.^$*[]')
                let all_commented = 0
                break
            endif
        endif
    endfor


    let result = []

    for line in lines

        let indent = matchstr(line, '^\s*')
        let text = substitute(line, '^\s*', '', '')

        if text ==# ''
            call add(result, line)
            continue
        endif


        if all_commented

            let text = substitute(
                \ text,
                \ '^' . escape(left,'\/.*$^~[]') . '\s*',
                \ '',
                \ '')

            if right !=# ''
                let text = substitute(
                    \ text,
                    \ '\s*' . escape(right,'\/.*$^~[]') . '$',
                    \ '',
                    \ '')
            endif

        else

            let text = left . ' ' . text

            if right !=# ''
                let text .= ' ' . right
            endif

        endif


        call add(result, indent . text)

    endfor


    call setline(a:start, result)

endfunction


nnoremap <silent> <leader>/ :
    \call ToggleComment(line('.'), line('.'))<CR>

xnoremap <silent> <leader>/ :
    \<C-u>call ToggleComment(line("'<"), line("'>"))<CR>

    autocmd!

    "----------------------------------------------
    " Restore cursor position
    "----------------------------------------------

    autocmd BufReadPost *
        \ if line("'\"") > 0 &&
        \ line("'\"") <= line("$") |
        \ execute "normal! g`\"" |
        \ endif

    "----------------------------------------------
    " Equalize splits after resize
    "----------------------------------------------

    autocmd VimResized *
        \ wincmd =

    "----------------------------------------------
    " Check for changed files
    "----------------------------------------------

    autocmd FocusGained,BufEnter *
        \ checktime

    "----------------------------------------------
    " Create parent directories automatically
    "----------------------------------------------

    autocmd BufWritePre *
        \ if !isdirectory(expand('%:p:h')) |
        \ call mkdir(expand('%:p:h'), 'p') |
        \ endif

    "----------------------------------------------
    " Disable heavy features for huge files
    "----------------------------------------------

    autocmd BufReadPre *
        \ if getfsize(expand('%')) > 1024 * 1024 |
        \ setlocal syntax=off |
        \ setlocal foldmethod=manual |
        \ setlocal noundofile |
        \ endif

    "----------------------------------------------
    " Highlight yanks
    "----------------------------------------------

    if exists('##TextYankPost')

        autocmd TextYankPost *
            \ silent! execute "normal! \<Esc>"

    endif


augroup END
